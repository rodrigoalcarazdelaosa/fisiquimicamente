{{- $html := string .html -}} {{- $replacements := dict -}} {{- $annotation := `([^<]*)` -}} {{- if .placeholder -}} {{/* Goldmark can turn paired carets in a shortcode's annotation into . */}} {{- $annotation = `((?:[^<]|)*)` -}} {{- end -}} {{- range $index, $math := uniq (findRE `(?s)]*>.*?` $html) -}} {{- range findRESubmatch $annotation $math 1 -}} {{- $display := findRE `^]* display="block"` $math 1 -}} {{- $delimiter := cond (ne (len $display) 0) "$$" "$" -}} {{- $tex := printf "%s%s%s" $delimiter (index . 1) $delimiter -}} {{- with $.placeholder -}} {{- $token := printf "%s%dZ" . $index -}} {{- $replacements = merge $replacements (dict $token ($tex | replaceRE `` "^" | htmlUnescape)) -}} {{- $tex = $token -}} {{- end -}} {{- $html = replace $html $math $tex -}} {{- end -}} {{- end -}} {{- return (dict "html" $html "replacements" $replacements) -}}
